ДЗ Скрипт
Напишите одну функцию на Bash, которая принимает путь к 'access.log' и выводит топ-5 IP по числу запросов. Используйте внутри цикла по массиву строк (прочитайте файл в массив через readarray) и ассоциативный массив для подсчёта. Сравните результат с однострочником из задачек, что эффективней и почему.

#!/usr/bin/env bash
top5_from_log() {
  local logfile="$1"
  if [[ ! -f "$logfile" ]]; then
    echo "File not found: $logfile" >&2
    return 1
  fi

  declare -A counts=()
  local -a lines
  readarray -t lines < "$logfile"

  local line ip
  for line in "${lines[@]}"; do
    ip=${line%% *}
    if [[ -n "$ip" ]]; then
      ((counts["$ip"]++))
    fi
  done

  for ip in "${!counts[@]}"; do
    printf "%d\t%s\n" "${counts[$ip]}" "$ip"
  done | sort -rn | head -n 5
}

# если вызван напрямую — выполнить
if [[ "${BASH_SOURCE[0]}" == "$0" ]]; then
  top5_from_log "${1:-access.log}"
fi



Однострочник (быстрее и экономнее по памяти на больших файлах):


awk '{print $1}' access.log | sort | uniq -c | sort -rn | head -n 5

Короткое сравнение эффективности:

Однострочник читает и обрабатывает по строчке, гораздо быстрее и менее требователен по памяти на больших файлах.
Функция с readarray + ассоциативный массив: загружает весь файл в память. Bash-цикл значительно медленнее
